# PowerShell como administrador. Atualiza o agente e o helper da Lixeira, sem recadastro.
& {
    $ErrorActionPreference = 'Stop'
    [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
    $arquivo = Join-Path $env:TEMP ("atualizar-infra-" + [guid]::NewGuid() + ".ps1")
    try {
        Invoke-WebRequest -UseBasicParsing -Uri 'https://raw.githubusercontent.com/fromNexz/UFFA/v2.1.1/maintenance/Atualizar-InfraMonitor-Web.ps1' -OutFile $arquivo
        if ((Get-FileHash $arquivo -Algorithm SHA256).Hash -ne 'edab066367f95b17d20883f7cbe61ca3cd19699f2fe53b73b852969d00a74b77') { throw 'Integridade invalida. Atualizacao cancelada.' }
        Unblock-File $arquivo
        powershell.exe -NoProfile -ExecutionPolicy RemoteSigned -File $arquivo
        if ($LASTEXITCODE -ne 0) { throw 'Atualizacao interrompida; confira a mensagem acima.' }
    } finally { Remove-Item $arquivo -Force -ErrorAction SilentlyContinue }
}
